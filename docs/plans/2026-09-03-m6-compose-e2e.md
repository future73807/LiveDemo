**Files:**

- [x] **Step 1: Dockerfile**

```dockerfile
FROM node:20-alpine AS build
WORKDIR /src
COPY package.json package-lock.json* ./
RUN npm install
COPY . .
RUN npm run build

FROM nginx:alpine
COPY --from=build /src/dist /usr/share/nginx/html
COPY nginx.conf /etc/nginx/conf.d/default.conf
EXPOSE 80
```

- [x] **Step 2: nginx.conf（含 WebSocket Upgrade 反代）**

```nginx
server {
    listen 80;

    location /api/ {
        proxy_pass http://live-service:8081;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    # 浏览器原生 WebSocket 走同源 /ws；gateway 模式下在此注入身份头（设计 §5.5）
    location /ws {
        proxy_pass http://live-service:8081;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_read_timeout 3600s;
    }

    location / {
        root /usr/share/nginx/html;
        try_files $uri /index.html;
    }
}
```

import { test, expect, type Page } from '@playwright/test';

async function login(page: Page, userId: string, nickname: string, role: 'VIEWER' | 'HOST' | 'ADMIN') {
  await page.goto('/');
  await page.getByPlaceholder('如 host1 / v1').fill(userId);
  await page.getByPlaceholder('直播间展示名').fill(nickname);
  await page.locator('select').selectOption(role);
  await page.getByRole('button', { name: '进入' }).click();
  await expect(page.getByText(new RegExp(nickname))).toBeVisible();
}

test('登录-建房-双端弹幕闭环', async ({ browser }) => {
  const hostPage = await browser.newContext().then(c => c.newPage());
  const viewerPage = await browser.newContext().then(c => c.newPage());

  // 主播：登录并建房
  await login(hostPage, 'e2e-host', 'E2E主播', 'HOST');
  await hostPage.getByRole('button', { name: '创建房间' }).click();
  await hostPage.getByPlaceholder('房间标题').fill('E2E 冒烟间');
  await hostPage.getByRole('button', { name: '创建', exact: true }).click();
  await hostPage.getByRole('button', { name: '进入直播间' }).click();
  await expect(hostPage.locator('.player-box')).toBeVisible();

  // 观众：从首页点进同一房间
  await login(viewerPage, 'e2e-viewer', 'E2E观众', 'VIEWER');
  await viewerPage.locator('.card', { hasText: 'E2E 冒烟间' }).first().click();

  // 观众发弹幕 → 双端可见（弹幕链路与推流解耦）
  const text = `hello-${Date.now()}`;
  await viewerPage.getByPlaceholder('发个弹幕…').fill(text);
  await viewerPage.getByRole('button', { name: '发送' }).click();
  await expect(hostPage.locator('.chat-item', { hasText: text })).toBeVisible({ timeout: 10_000 });
  await expect(viewerPage.locator('.chat-item', { hasText: text })).toBeVisible();
});

test('管理员后台可见强制关播按钮', async ({ page }) => {
  await login(page, 'e2e-admin', 'E2E管理员', 'ADMIN');
  await page.getByRole('link', { name: '管理后台' }).click();
  await expect(page.getByRole('button', { name: '强制关播' }).first()).toBeVisible();

### 对计划的修改点（执行偏差）

1. **Task 1 增补 `web/.dockerignore`**（node_modules/dist/tests-e2e/playwright 产物）：否则 `COPY . .` 会把数百 MB 的本机 node_modules 打进构建上下文，属必要补充。
2. **Task 1 Step 3 单独验证改为挂 `livedemo_default` 网络运行**：nginx 在 upstream `live-service` 不可解析时无法启动（`host not found in upstream`），原计划的纯静态独立运行不成立；挂 compose 网络后同时顺带验证了反代链。
3. **Task 3 用例两处修正（以实际 UI 代码为准，未改前端文案）**：
   - 「房间标题」实际是 `<label>` 而非 placeholder → 改用 `.dialog .field` 定位输入框；
   - 登录成功断言限定顶栏（历史房间卡片会显示同昵称主播，全局 getByText 撞 Playwright strict mode）；
   - 房间标题带 `Date.now()` 时间戳：H2 数据卷持久化导致历史同名房间累积，固定标题会让观众点进旧房间、弹幕闭环失败。
4. **新增 `vite.config.ts` 的 vitest `include` 限定**：vitest 默认通配会误收集 `tests-e2e/*.spec.ts` 导致 `npm test` 失败。
5. **执行中发现并修复两个集成缺陷（M6 验证的价值所在）**：
   - `srs/srs.conf` 缺 `http_remux` 块，8080 的 `.flv` 返回 404（设计 §6.2 降级协议失效）→ 已补上并验证；
   - `RoomSocketHandler` 连接建立时只发 history/presence、不补发当前房间状态，中途进场的观众永远收不到 `room_status:LIVING`、播放器不拉流 → 已在连接时补发（含 47 例单测回归与无头浏览器 WHEP 出画验证）。
6. **e2e 增加一条轻量断言**（观众页 `N 人在线`），佐证清单第 3 项在线人数展示。
